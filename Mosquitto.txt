sudo bash -c 'cat <<EOF > /etc/mosquitto/conf.d/tuning.conf
listener 1883
allow_anonymous true

# Keep all broker state in RAM; avoid disk sync stalls
persistence false

# Disable disk logging so high-frequency simulator packets do not block I/O
log_dest none

# Remove queues and allow instant in-flight delivery
max_inflight_messages 0
max_queued_messages 0
message_size_limit 0
EOF'

sudo systemctl restart mosquito

python3 -c "
import paho.mqtt.client as mqtt, time
count = 0
start = time.time()
def on_msg(c, u, m):
    global count, start
    count += 1
    now = time.time()
    if now - start >= 1.0:
        print(f'Sensor loop rate: {count / (now - start):.1f} Hz')
        count, start = 0, now
c = mqtt.Client(mqtt.CallbackAPIVersion.VERSION2)
c.on_message = on_msg
c.connect('localhost', 1883)
c.subscribe('pacbot/sensors')
c.loop_forever()
"
